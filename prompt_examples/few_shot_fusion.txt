    design = app.activeProduct\
    root = design.rootComponent\
    sketch = root.sketches.add(root.xYConstructionPlane)\
    sketch.sketchCurves.sketchLines.addTwoPointRectangle(\

    )\

    extInput.setDistanceExtent(False, adsk.core.ValueInput.createByReal(0.2))\
    body = extrudes.add(extInput).bodies.item(0)\
\
    chamferFeats = root.features.chamferFeatures\
    edges = adsk.core.ObjectCollection.create()\
    for edge in body.edges:\
        edges.add(edge)\
    chamferInput = chamferFeats.createInput2()\
    chamferInput.chamferEdgeSets.addEqualDistanceChamferEdgeSet(edges, adsk.core.ValueInput.createByReal(0.1), True)\
    chamferFeats.add(chamferInput)\
# End\
